# AcadeXMatch settings. Create a file named ".env.local" in the project folder (next to package.json),
# copy the lines below into it and fill in your values, then restart the app (npm run dev / npm start).
# .env.local is private: never share or commit it.

# ---- Database ----
# On GoDaddy Node.js Hosting these are filled in automatically for the app's MySQL database
# (DB_HOST, DB_PORT, DB_NAME, DB_USER, DB_PASSWORD) — you do not need to add them there.
# Locally, leave them out to keep data in the "data" folder, or point them at your own MySQL server.
# DB_HOST=localhost
# DB_PORT=3306
# DB_NAME=acadexmatch
# DB_USER=acadexmatch
# DB_PASSWORD=your-password

# Public address of the site, used in links inside emails.
APP_URL=http://localhost:3000

# ---- Email (Gmail example) ----
# 1. Turn on 2-Step Verification for the Gmail account.
# 2. Create an App Password: Google Account → Security → App passwords.
# 3. Put the 16-character app password below (no spaces).
SMTP_HOST=smtp.gmail.com
SMTP_PORT=465
SMTP_SECURE=true
SMTP_USER=yourname@gmail.com
SMTP_PASS=your16charapppassword
MAIL_FROM=AcadeXMatch <yourname@gmail.com>

# ---- Email through an API (use this if your host blocks email ports 465/587) ----
# Some hosts (e.g. GoDaddy Node.js hosting) block outgoing SMTP. Then send over HTTPS instead:
#  Brevo (brevo.com, free 300 emails/day): Settings → SMTP & API → API keys → create a key;
#    Senders & domains → add info@acadexmatch.com as a sender and verify it (and the domain, recommended).
#  or Resend (resend.com): add and verify the domain acadexmatch.com, then create an API key.
# When EMAIL_API is set it is used instead of SMTP.
# EMAIL_API=brevo
# EMAIL_API_KEY=xkeysib-xxxxxxxx
# MAIL_FROM=AcadeXMatch <info@acadexmatch.com>

# ---- Razorpay (online payments) ----
# Razorpay Dashboard → Account & Settings → API Keys. Start with the Test Mode keys (rzp_test_...);
# switch to the Live keys after Razorpay activates your account. Leave these out to keep the
# manual "request upgrade" flow instead of online payment.
RAZORPAY_KEY_ID=rzp_test_xxxxxxxxxxxx
RAZORPAY_KEY_SECRET=xxxxxxxxxxxxxxxxxxxx
# Optional but recommended once the site has a public https address:
# Dashboard → Webhooks → Add: URL https://www.acadexmatch.com/api/payments/webhook, events
# payment.captured, order.paid, payment.failed, refund.created, refund.processed, refund.failed,
# and the same secret here. It confirms payments even if the buyer closes the browser and records refunds.
# RAZORPAY_WEBHOOK_SECRET=choose-a-long-random-secret

# Optional: set to 1 to stop the daily vacancy scrape from running inside the web server.
# ACAD_DISABLE_SCHEDULER=1
